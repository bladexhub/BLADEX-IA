# ========================================================
# Script de Limpieza Total: Discord y Telegram Desktop
# ========================================================

# 1. Verificar si se ejecuta como Administrador
if (!([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "ERROR: Debes ejecutar este script como Administrador." -ForegroundColor Red
    Write-Host "Cierra esta ventana, haz clic derecho en el archivo .ps1 y selecciona 'Ejecutar como administrador'." -ForegroundColor Yellow
    Pause
    Exit
}

Write-Host "Iniciando limpieza profunda de Discord y Telegram..." -ForegroundColor Cyan
Write-Host "========================================================"

# 2. Cerrar procesos de Discord y Telegram
Write-Host "`n[1/5] Cerrando procesos en segundo plano..." -ForegroundColor Cyan
Get-Process -Name "Discord", "Telegram" -ErrorAction SilentlyContinue | Stop-Process -Force
Write-Host "[OK] Procesos cerrados (si estaban abiertos)." -ForegroundColor Green

# 3. Eliminar carpetas de instalación y datos residuales
Write-Host "`n[2/5] Eliminando carpetas de instalación y datos residuales..." -ForegroundColor Cyan
$carpetas = @(
    "$env:LOCALAPPDATA\Discord",
    "$env:APPDATA\Discord",
    "$env:APPDATA\Telegram Desktop"
)
foreach ($carpeta in $carpetas) {
    if (Test-Path $carpeta) {
        Remove-Item -Path $carpeta -Recurse -Force -ErrorAction SilentlyContinue
        Write-Host "[OK] Eliminado: $carpeta" -ForegroundColor Green
    }
}

# 4. Eliminar accesos del Menú de Inicio
Write-Host "`n[3/5] Eliminando accesos del Menú de Inicio..." -ForegroundColor Cyan
$menus = @(
    "$env:APPDATA\Microsoft\Windows\Start Menu\Programs\Discord Inc",
    "$env:APPDATA\Microsoft\Windows\Start Menu\Programs\Telegram Desktop"
)
foreach ($menu in $menus) {
    if (Test-Path $menu) {
        Remove-Item -Path $menu -Recurse -Force -ErrorAction SilentlyContinue
        Write-Host "[OK] Menú Inicio limpiado: $menu" -ForegroundColor Green
    }
}

# 5. Limpiar Registro de Windows
Write-Host "`n[4/5] Limpiando Registro de Windows..." -ForegroundColor Cyan
$registros = @(
    "HKCU:\Software\Discord",
    "HKCU:\Software\Microsoft\Windows\CurrentVersion\Uninstall\Discord",
    "HKCU:\Software\TelegramDesktop",
    "HKCU:\Software\Microsoft\Windows\CurrentVersion\Uninstall\{53F49750-6209-4FBF-9CA8-7A333C87D1ED}_is1"
)
foreach ($reg in $registros) {
    if (Test-Path $reg) {
        Remove-Item -Path $reg -Recurse -Force -ErrorAction SilentlyContinue
        Write-Host "[OK] Registro limpiado: $reg" -ForegroundColor Green
    }
}

# 6. Reiniciar el Explorador de Windows
Write-Host "`n[5/5] Reiniciando el Explorador de Windows..." -ForegroundColor Cyan
Stop-Process -Name "explorer" -Force -ErrorAction SilentlyContinue
Start-Process "explorer.exe"
Write-Host "[OK] Explorador reiniciado." -ForegroundColor Green

Write-Host "`n========================================================"
Write-Host "¡Limpieza completada! Discord y Telegram han sido eliminados." -ForegroundColor Green
Write-Host "========================================================"
Write-Host "`nPresiona cualquier tecla para salir..."
$Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown") | Out-Null
