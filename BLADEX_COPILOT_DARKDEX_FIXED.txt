-- inspect the currently running DataModel. Remote AI requires a Gemini API key and
-- an HTTP request function supplied by the execution environment.

-- ============================================================
-- Compatibility / services
-- ============================================================

local function safe(f, ...)
    local ok, a, b, c = pcall(f, ...)
    if ok then return a, b, c end
    return nil, a
end

local HttpService = safe(function() return game:GetService("HttpService") end)
local Players = safe(function() return game:GetService("Players") end)
local CoreGui = safe(function() return game:GetService("CoreGui") end)
local UserInputService = safe(function() return game:GetService("UserInputService") end)

if not game or not HttpService or not Players then
    error("BLADEX Copilot: Roblox services are unavailable.")
end

local function instancePath(inst)
    if not inst then return "nil" end

    local parts = {}
    local cur = inst
    local guard = 0

    while cur and cur ~= game and guard < 80 do
        guard += 1
        table.insert(parts, 1, cur.Name)
        cur = cur.Parent
    end

    if cur == game then
        return "game." .. table.concat(parts, ".")
    end

    return table.concat(parts, ".")
end

local function resolvePath(path)
    if type(path) ~= "string" or path == "" then
        return nil, "empty path"
    end

    if path == "game" then return game end
    if path == "workspace" then return workspace end

    local cur = game
    local text = path

    if text:sub(1, 5) == "game." then
        text = text:sub(6)
    elseif text:sub(1, 10) == "workspace." then
        cur = workspace
        text = text:sub(11)
    end

    for segment in string.gmatch(text, "([^%.]+)") do
        local child = safe(function()
            return cur:FindFirstChild(segment)
        end)

        if not child and cur == game then
            child = safe(function()
                return game:GetService(segment)
            end)
        end

        if not child then
            return nil, "not found: " .. segment
        end

        cur = child
    end

    return cur
end

local function serialize(value, depth)
    depth = depth or 0
    if depth > 3 then return "<depth-limit>" end

    local kind = typeof(value)

    if kind == "Instance" then
        return instancePath(value)
    elseif kind == "Vector3" then
        return {x=value.X, y=value.Y, z=value.Z}
    elseif kind == "Vector2" then
        return {x=value.X, y=value.Y}
    elseif kind == "CFrame" then
        local p = value.Position
        return {x=p.X, y=p.Y, z=p.Z}
    elseif kind == "Color3" then
        return {r=value.R, g=value.G, b=value.B}
    elseif kind == "EnumItem" then
        return tostring(value)
    elseif kind == "table" then
        local out = {}
        local n = 0
        for k,v in pairs(value) do
            n += 1
            if n > 40 then
                out["..."] = "truncated"
                break
            end
            out[tostring(k)] = serialize(v, depth + 1)
        end
        return out
    end

    if type(value) == "number" or type(value) == "string" or type(value) == "boolean" then
        return value
    end

    return tostring(value)
end

local function listChildren(path)
    local inst, err = resolvePath(path or "game")
    if not inst then return {error=err} end

    local result = {
        path = instancePath(inst),
        name = inst.Name,
        class = inst.ClassName,
        children = {},
    }

    local children = safe(function() return inst:GetChildren() end) or {}
    table.sort(children, function(a,b) return a.Name:lower() < b.Name:lower() end)

    for i, child in ipairs(children) do
        if i > Config.MaxChildren then
            result.truncated = true
            break
        end

        table.insert(result.children, {
            name = child.Name,
            class = child.ClassName,
            path = instancePath(child),
        })
    end

    return result
end

local function getProperties(path, names)
    local inst, err = resolvePath(path)
    if not inst then return {error=err} end

    local wanted = {}
    if type(names) == "table" then
        for _,n in ipairs(names) do wanted[tostring(n)] = true end
    end

    local defaults = {
        "Name", "ClassName", "Parent", "Archivable",
        "Position", "Size", "CFrame", "Anchored",
        "CanCollide", "Transparency", "Visible",
        "Enabled", "Value", "Text", "DisplayName",
        "Health", "MaxHealth", "WalkSpeed", "JumpPower",
    }

    local result = {
        path = instancePath(inst),
        class = inst.ClassName,
        properties = {},
    }

    local list = #names > 0 and names or defaults

    for _, property in ipairs(list) do
        if property ~= "ClassName" then
            local ok, value = pcall(function()
                return inst[property]
            end)
            if ok then
                result.properties[property] = serialize(value)
            end
        end
    end

    result.properties.ClassName = inst.ClassName
    return result
end

local function getAttributes(path)
    local inst, err = resolvePath(path)
    if not inst then return {error=err} end

    local ok, attrs = pcall(function() return inst:GetAttributes() end)
    if not ok then return {error=tostring(attrs)} end

    return {
        path = instancePath(inst),
        attributes = serialize(attrs),
    }
end

local function getCFrame(path)
    local inst, err = resolvePath(path)
    if not inst then return {error=err} end

    local ok, cf = pcall(function() return inst.CFrame end)
    if not ok then return {error="instance has no readable CFrame"} end

    return {
        path = instancePath(inst),
        cframe = serialize(cf),
    }
end

local function searchInstances(query, className)
    query = tostring(query or ""):lower()
    if query == "" then return {error="empty search query"} end

    local results = {}
    local descendants = safe(function() return game:GetDescendants() end) or {}

    for _, inst in ipairs(descendants) do
        if #results >= Config.MaxSearchResults then break end

        local nameMatch = string.find(inst.Name:lower(), query, 1, true) ~= nil
        local classMatch = (not className or className == "" or inst.ClassName == className)

        if nameMatch and classMatch then
            table.insert(results, {
                name = inst.Name,
                class = inst.ClassName,
                path = instancePath(inst),
            })
        end
    end

    return {
        query = query,
        count = #results,
        results = results,
        truncated = #results >= Config.MaxSearchResults,
    }
end

local function getSelected()
    -- Optional integration with common selection globals.
    local selected =
        G.BLADEX_SELECTED_INSTANCE or
        G.SelectedInstance or
        G.Selected

    if typeof(selected) == "Instance" then
        return {
            path = instancePath(selected),
            name = selected.Name,
            class = selected.ClassName,
        }
    end

    return {error="no compatible selection provider is available"}
end

local function getGameInfo()
    local info = {
        place_id = game.PlaceId,
        game_id = game.GameId,
        job_id = game.JobId,
        place_name = "unknown",
        creator = "unknown",
        player_count = #Players:GetPlayers(),
    }

    local MarketplaceService = safe(function()
        return game:GetService("MarketplaceService")
    end)

    if MarketplaceService then
        local ok, data = pcall(function()
            return MarketplaceService:GetProductInfo(game.PlaceId)
        end)

        if ok and type(data) == "table" then
            info.place_name = data.Name or info.place_name
            info.creator = data.Creator and (data.Creator.Name or tostring(data.Creator.Id)) or info.creator
            info.description = data.Description
        end
    end

    return info
end

local function getLocalPlayer()
    if not LocalPlayer then return {error="LocalPlayer unavailable"} end

    local out = {
        name = LocalPlayer.Name,
        display_name = LocalPlayer.DisplayName,
        user_id = LocalPlayer.UserId,
        team = nil,
    }

    pcall(function()
        if LocalPlayer.Team then out.team = LocalPlayer.Team.Name end
    end)

    local character = LocalPlayer.Character
    if character then
        out.character = instancePath(character)

        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            out.health = humanoid.Health
            out.max_health = humanoid.MaxHealth
            out.walk_speed = humanoid.WalkSpeed
            out.jump_power = humanoid.JumpPower
        end

        local root = character:FindFirstChild("HumanoidRootPart")
        if root then
            out.position = serialize(root.Position)
        end
    end

    return out
end

-- Script source is executor-dependent. We only use read-only providers if available.
local function getScriptSource(path)
    local inst, err = resolvePath(path)
    if not inst then return {error=err} end

    local supported =
        inst:IsA("LocalScript") or
        inst:IsA("ModuleScript") or
        inst:IsA("Script")

    if not supported then
        return {error="target is not a Script, LocalScript or ModuleScript"}
    end

    local providers = {
        G.getscript,
        G.decompile,
        G.get_script_source,
    }

    for _, provider in ipairs(providers) do
        if type(provider) == "function" then
            local ok, source = pcall(provider, inst)
            if ok and type(source) == "string" and #source > 0 then
                if #source > Config.MaxScriptChars then
                    source = source:sub(1, Config.MaxScriptChars) ..
                        "\n-- [truncated by BLADEX Copilot]"
                end

                return {
                    path = instancePath(inst),
                    class = inst.ClassName,
                    source = source,
                }
            end
        end
    end

    return {
        path = instancePath(inst),
        class = inst.ClassName,
        error="No compatible read-only script-source provider is exposed by this environment.",
    }
end

local SystemPrompt = [[
You are BLADEX Copilot, a read-only Roblox runtime inspection assistant.

You are connected to the currently running DataModel. Use the supplied tools instead
of guessing. Inspect progressively: start with the smallest useful query, then chain
additional read-only calls when necessary.

You can inspect:
- DataModel hierarchy
- instance names/classes
- selected properties and attributes
- CFrames
- name-based searches
- game metadata
- local-player runtime information
- script source only when the execution environment exposes a read-only provider

Never claim that a value was inspected if a tool did not return it.
Keep tool arguments small and focused.
Reply in the user's language.
]]

local GEMINI_MODELS = {
    "gemini-3.8-flash",
    "gemini-3.7-flash",
    "gemini-3.6-flash",
    "gemini-3.5-flash",
    "gemini-3.5-flash-lite",
}

local function normalizeResponse(response)
    if type(response) ~= "table" then
        return 0, ""
    end
    local status = tonumber(response.StatusCode or response.Status or response.status_code or 0) or 0
    local raw = response.Body or response.body or response.ResponseBody or ""
    return status, tostring(raw)
end

local function requestGeminiWithModel(model, contents)
    local url =
        "https://generativelanguage.googleapis.com/v1beta/models/" ..
        tostring(model) ..
        ":generateContent?key=" ..
        tostring(Config.ApiKey)

    local body = {
        contents = contents,
        systemInstruction = {parts = {{text = SystemPrompt}}},
        tools = toolDeclarations(),
        generationConfig = {
            maxOutputTokens = Config.MaxOutputTokens,
        },
    }

    local encoded = HttpService:JSONEncode(body)

    local ok, response = pcall(REQUEST, {
        Url = url,
        Method = "POST",
        Headers = {
            ["Content-Type"] = "application/json",
            ["Accept"] = "application/json",
        },
        Body = encoded,
    })

    if not ok then
        return nil, "HTTP request failed: " .. tostring(response), 0
    end

    local status, raw = normalizeResponse(response)

    if status >= 400 then
        return nil, raw:sub(1, 500), status
    end

    local decodeOk, decoded = pcall(function()
        return HttpService:JSONDecode(raw)
    end)

    if not decodeOk or type(decoded) ~= "table" then
        return nil, "The API returned invalid JSON.", status
    end

    if decoded.error then
        local msg = decoded.error.message or tostring(decoded.error)
        return nil, tostring(msg), status
    end

    return decoded, nil, status
end

local function requestGemini(contents)
    if type(REQUEST) ~= "function" then
        return nil, "HTTP request is not available in this environment."
    end

    if Config.ApiKey == "" then
        return nil, "Gemini API key is empty."
    end

    local first = tostring(Config.Model or GEMINI_MODELS[1])
    local candidates = {first}

    for _, model in ipairs(GEMINI_MODELS) do
        if model ~= first then
            table.insert(candidates, model)
        end
    end

    local lastError = "Unknown Gemini error."

    for _, model in ipairs(candidates) do
        local decoded, err, status = requestGeminiWithModel(model, contents)

        if decoded then
            Config.Model = model
            return decoded
        end

        lastError = "HTTP " .. tostring(status) .. ": " .. tostring(err)

        if status ~= 404 then
            return nil, lastError
        end
    end

    return nil, "No available Gemini model was found. Last error: " .. lastError
end

-- Context fallback

local function buildLocalContext(message)
    local lower = tostring(message):lower()
    local context = {}

    table.insert(context, {
        type="game_info",
        data=getGameInfo(),
    })

    -- Selected-instance requests
    if lower:find("selected",1,true) or lower:find("seleccion",1,true) then
        table.insert(context, {
            type="selected",
            data=getSelected(),
        })
    end

    -- Common hierarchy questions
    if lower:find("workspace",1,true) or
       lower:find("children",1,true) or
       lower:find("hijos",1,true) then
        table.insert(context, {
            type="workspace",
            data=listChildren("game.Workspace"),
        })
    end

    if lower:find("player",1,true) or lower:find("jugador",1,true) then
        table.insert(context, {
            type="local_player",
            data=getLocalPlayer(),
        })
    end

    return context
end

-- ============================================================
-- Minimal markdown renderer
-- ============================================================

local function escapeRichText(s)
    s = tostring(s or "")
    s = s:gsub("&", "&amp;")
    s = s:gsub("<", "&lt;")
    s = s:gsub(">", "&gt;")
    return s
end

local function markdown(text)
    local s = escapeRichText(text)
    s = s:gsub("```[%w_%-]*\n(.-)```", "<font face=\"Code\">%1</font>")
    s = s:gsub("```(.-)```", "<font face=\"Code\">%1</font>")
    s = s:gsub("`([^`\n]+)`", "<font face=\"Code\">%1</font>")
    s = s:gsub("%*%*(.-)%*%*", "<b>%1</b>")
    s = s:gsub("%*([^%*\n]+)%*", "<i>%1</i>")
    return s
end

local function cleanUiText(text)
    text = tostring(text or "")
    text = text:gsub("\226\128\166", "...")
    text = text:gsub("\226\128\147", "-")
    text = text:gsub("\226\128\148", "-")
    text = text:gsub("\226\128\152", "'")
    text = text:gsub("\226\128\153", "'")
    text = text:gsub("\226\128\156", "\"")
    text = text:gsub("\226\128\157", "\"")
    text = text:gsub("\226\128\162", "-")
    text = text:gsub("\194\160", " ")
    return text
end

local function addMessage(kind, text)
    if not State.Chat then return end

    text = cleanUiText(text)

    local row = Instance.new("Frame")
    row.Name = kind == "user" and "UserMessage" or "CopilotMessage"
    row.BackgroundTransparency = 1
    row.BorderSizePixel = 0
    row.Size = UDim2.new(1, -16, 0, 0)
    row.AutomaticSize = Enum.AutomaticSize.Y
    row.Parent = State.Chat

    local bubble = Instance.new("Frame")
    bubble.BorderSizePixel = 0
    bubble.AutomaticSize = Enum.AutomaticSize.Y
    bubble.Size = UDim2.new(0.78, 0, 0, 0)
    bubble.Parent = row

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 14)
    corner.Parent = bubble

    local stroke = Instance.new("UIStroke")
    stroke.Thickness = 1
    stroke.Transparency = 0.55
    stroke.Parent = bubble

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.BorderSizePixel = 0
    label.Font = Enum.Font.Gotham
    label.TextSize = 13
    label.TextWrapped = true
    label.RichText = kind == "bot"
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextYAlignment = Enum.TextYAlignment.Top
    label.AutomaticSize = Enum.AutomaticSize.Y
    label.Size = UDim2.new(1, -24, 0, 0)
    label.Position = UDim2.new(0, 12, 0, 9)
    label.Text = kind == "bot" and markdown(text) or escapeRichText(text)
    label.Parent = bubble

    local padding = Instance.new("UIPadding")
    padding.PaddingBottom = UDim.new(0, 9)
    padding.Parent = label

    if kind == "user" then
        bubble.AnchorPoint = Vector2.new(1, 0)
        bubble.Position = UDim2.new(1, 0, 0, 0)
        bubble.BackgroundColor3 = Color3.fromRGB(10, 132, 255)
        stroke.Color = Color3.fromRGB(55, 160, 255)
        label.TextColor3 = Color3.fromRGB(255, 255, 255)
    else
        bubble.AnchorPoint = Vector2.new(0, 0)
        bubble.Position = UDim2.new(0, 0, 0, 0)
        bubble.BackgroundColor3 = Color3.fromRGB(36, 36, 44)
        stroke.Color = Color3.fromRGB(70, 70, 82)
        label.TextColor3 = Color3.fromRGB(236, 236, 240)
    end

    task.defer(function()
        if State.Chat then
            State.Chat.CanvasPosition =
                Vector2.new(0, math.max(0, State.Chat.AbsoluteCanvasSize.Y))
        end
    end)
end

local function clearChat()
    State.History = {}
    State.Messages = {}

    if State.Chat then
        for _, child in ipairs(State.Chat:GetChildren()) do
            if child:IsA("Frame") then
                child:Destroy()
            end
        end
    end
end

local function destroy()
    State.Destroyed = true
    if State.Gui then
        State.Gui:Destroy()
    end
    G.BLADEX_COPILOT = nil
end

-- ============================================================
-- Agent loop
-- ============================================================

local function runAgent(message)
    if State.Busy then return end

    State.Busy = true
    setStatus("Thinking...")

    if Config.ApiKey == "" then
        addMessage("bot",
            "Warning: Configura tu Gemini API key en el campo superior.\n" ..
            "La inspeccion local del juego funciona sin API; la conversacion con el modelo requiere la clave.")
        State.Busy = false
        setStatus("")
        return
    end

    table.insert(State.History, {
        role="user",
        parts={{text=message}},
    })

    local localContext = buildLocalContext(message)
    if #localContext > 0 then
        table.insert(State.History, {
            role="user",
            parts={{
                text="Local runtime context:\n" ..
                    HttpService:JSONEncode(localContext),
            }},
        })
    end

    for iteration=1,Config.MaxIterations do
        if State.Destroyed then break end

        setStatus("Thinking " .. tostring(iteration) .. "/" .. tostring(Config.MaxIterations) .. "...")

        local decoded, err = requestGemini(State.History)
        if not decoded then
            addMessage("bot", "Warning: " .. tostring(err))
            break
        end

        local candidate = decoded.candidates and decoded.candidates[1]
        local parts = candidate and candidate.content and candidate.content.parts

        if type(parts) ~= "table" then
            addMessage("bot", "Warning: Gemini returned an empty response.")
            break
        end

        local textOut = {}
        local calls = {}

        for _, part in ipairs(parts) do
            if part.text then
                table.insert(textOut, part.text)
            end

            if part.functionCall then
                table.insert(calls, part.functionCall)
            end
        end

        table.insert(State.History, {
            role="model",
            parts=parts,
        })

        if #textOut > 0 then
            addMessage("bot", table.concat(textOut,"\n"))
        end

        if #calls == 0 then
            break
        end

        local responseParts = {}

        for _, call in ipairs(calls) do
            local name = call.name
            local args = call.args or {}
            local tool = Tools[name]

            addMessage("tool",
                "> " .. tostring(name) .. " " ..
                HttpService:JSONEncode(args):sub(1,300))

            setStatus("Inspecting: " .. tostring(name))

            local result

            if not tool then
                result = {error="unknown tool: " .. tostring(name)}
            else
                local ok, output = pcall(tool.run, args)
                if ok then
                    result = output
                else
                    result = {error=tostring(output)}
                end
            end

            table.insert(responseParts, {
                functionResponse={
                    name=name,
                    response={content=result},
                },
            })
        end

        table.insert(State.History, {
            role="user",
            parts=responseParts,
        })
    end

    State.Busy = false
    setStatus("")
end

local function send()
    if State.Busy or State.Destroyed or not State.Input then return end

    local text = State.Input.Text
    if not text or text:gsub("%s","") == "" then return end

    State.Input.Text = ""
    addMessage("user", text)

    task.spawn(function()
        runAgent(text)
    end)
end
